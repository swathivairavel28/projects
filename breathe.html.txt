    <div id="circle" class="circle">Inhale</div>
    <br />
    <button onclick="resetBreathe()">Reset</button>
  </div>

  <script>
    const circle = document.getElementById("circle");
    let step = 0;
    const phases = ["Inhale", "Hold", "Exhale"];

    function breatheCycle() {
      circle.classList.add("grow");
      circle.textContent = phases[step];

      setTimeout(() => {
        circle.classList.remove("grow");
        step = (step + 1) % phases.length;
        setTimeout(breatheCycle, 1000); // pause before next cycle
      }, 5000);
    }

    function resetBreathe() {
      step = 0;
      circle.textContent = phases[step];
    }

    breatheCycle();
  </script>
